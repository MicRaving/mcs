# ---- replace the single handler with these three (inside the same site block, ---
# ---- after `abort @meta_public`; the @meta / @meta_public matchers stay) --------

	# the browser app: the gateway owns the session and proxies to a per-user worker
	@meta_browser {
		host meta.home.mfendt.de
		path /app /app/*
	}
	handle @meta_browser {
		request_body {
			max_size 250MB
		}
		reverse_proxy 192.168.178.12:8780
	}

	# everything else: the sync server (desktop clients, unchanged)
	handle @meta {
		request_body {
			max_size 250MB
		}
		reverse_proxy 192.168.178.12:8770
	}

	# a bare "/" is not the app any more: send the browser to the gateway
	@meta_root {
		host meta.home.mfendt.de
		path /
	}
	redir @meta_root /app/ 302

# ---------------------------------------------------------------------------
# Why `path /app /app/*`: Caddy's path matcher is exact for /app and prefix for
# /app/*, so both the bare path (the gateway redirects it to /app/) and every
# route below it reach the gateway.  `redir` is ordered before `handle`, so a
# request for "/" is answered before either backend is consulted.
